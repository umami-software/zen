A switch allows a user to turn a setting on or off.

<Example>
  <Switch>Low power mode</Switch>

  ```jsx
  <Switch>Low power mode</Switch>
  ```
</Example>

## Disabled

<Example>
  <Switch isDisabled>Disabled off</Switch>
  <Switch defaultSelected={true} isDisabled>
    Disabled on
  </Switch>

  ```jsx
  <Switch isDisabled>Disabled off</Switch>
  <Switch defaultSelected isDisabled>Disabled on</Switch>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text rendered next to the switch (`children` works the same way)</ResponseField>
  <ResponseField name="size" type="'sm' | 'md'">Track size (default: "md")</ResponseField>
  <ResponseField name="isSelected" type="boolean">Controlled selected state</ResponseField>
  <ResponseField name="defaultSelected" type="boolean">Default selected state</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the switch</ResponseField>
  <ResponseField name="onChange" type="(isSelected: boolean) => void">Called when state changes</ResponseField>
  <ResponseField name="value" type="string | boolean">A string is forwarded to the form input as the submitted value. A boolean is a deprecated alias for `isSelected` — prefer `isSelected`.</ResponseField>
</PropertiesTable>

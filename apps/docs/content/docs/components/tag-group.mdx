A tag group displays a collection of tags that can optionally be removed.

<Example>
  <TagGroup label="Categories">
    <Tag id="react">React</Tag>
    <Tag id="vue">Vue</Tag>
    <Tag id="angular">Angular</Tag>
  </TagGroup>

  ```jsx
  <TagGroup label="Categories">
    <Tag id="react">React</Tag>
    <Tag id="vue">Vue</Tag>
    <Tag id="angular">Angular</Tag>
  </TagGroup>
  ```
</Example>

## Removable

Add `onRemove` to make tags removable. An X button will appear on each tag.

<Example>
  <RemovableTagGroupExample />

  ```jsx
  <TagGroup label="Skills" onRemove={keys => console.log('Remove:', [...keys])}>
    <Tag id="javascript">JavaScript</Tag>
    <Tag id="typescript">TypeScript</Tag>
    <Tag id="python">Python</Tag>
  </TagGroup>
  ```
</Example>

## Component props

### TagGroup

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the group</ResponseField>
  <ResponseField name="onRemove" type="(keys: Set<Key>) => void">Called when tags are removed. Enables remove buttons.</ResponseField>
  <ResponseField name="disabledKeys" type="Iterable<Key>">Keys of tags that cannot be removed</ResponseField>
</PropertiesTable>

### Tag

<PropertiesTable>
  <ResponseField name="id" type="string">Unique tag identifier (required)</ResponseField>
  <ResponseField name="variant" type="&quot;default&quot; | &quot;outline&quot; | &quot;primary&quot;">Visual style variant</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the tag</ResponseField>
</PropertiesTable>

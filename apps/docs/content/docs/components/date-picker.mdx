import { DatePicker, DatePickerExample, Example } from '../../../mdx-components';

A date picker allows a user to select a date from a popup calendar.

<Example>
  <DatePicker label="Date" />

  ```jsx
  <DatePicker label="Date" />
  ```
</Example>

## Default value

<Example>
  <DatePicker label="Date" defaultValue={new Date()} />

  ```jsx
  <DatePicker label="Date" defaultValue={new Date()} />
  ```
</Example>

## Format options

Dates are formatted with `Intl.DateTimeFormat` using the `locale` and `formatOptions` props.

<Example>
  <DatePicker label="Date" defaultValue={new Date()} formatOptions={{ dateStyle: 'full' }} />

  ```jsx
  <DatePicker label="Date" defaultValue={new Date()} formatOptions={{ dateStyle: 'full' }} />
  ```
</Example>

## Disabled

<Example>
  <DatePicker label="Date" isDisabled />

  ```jsx
  <DatePicker label="Date" isDisabled />
  ```
</Example>

## Controlled

<Example>
  <DatePickerExample />

  ```jsx
  const [date, setDate] = useState();

  <DatePicker label="Date" value={date} onChange={setDate} />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the picker</ResponseField>
  <ResponseField name="value" type="Date">Controlled date value</ResponseField>
  <ResponseField name="defaultValue" type="Date">Default date value</ResponseField>
  <ResponseField name="minValue" type="Date">Minimum selectable date</ResponseField>
  <ResponseField name="maxValue" type="Date">Maximum selectable date</ResponseField>
  <ResponseField name="placeholder" type="string">Text shown when no date is selected (default: "Select date")</ResponseField>
  <ResponseField name="locale" type="string">Locale used to format the displayed date</ResponseField>
  <ResponseField name="formatOptions" type="Intl.DateTimeFormatOptions">{"Date format options (default: { dateStyle: \"medium\" })"}</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the picker</ResponseField>
  <ResponseField name="isReadOnly" type="boolean">Prevents changing the value</ResponseField>
  <ResponseField name="onChange" type="(date: Date) => void">Called when a date is selected</ResponseField>
  <ResponseField name="buttonProps" type="ButtonProps">Props passed to the trigger button</ResponseField>
  <ResponseField name="calendarProps" type="Partial<CalendarProps>">Props passed to the popup calendar</ResponseField>
</PropertiesTable>

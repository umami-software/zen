import { DatePicker, DatePickerExample, Example } from '../../../mdx-components';

A date picker allows a user to select a date from a popup calendar. It combines the shadcn
`Popover`, `Calendar` and an outline `Button` trigger.

<Example>
  <DatePicker label="Date" />

  ```jsx
  <DatePicker label="Date" />
  ```
</Example>

## Default value

<Example>
  <DatePicker label="Date" defaultValue={new Date()} />

  ```jsx
  <DatePicker label="Date" defaultValue={new Date()} />
  ```
</Example>

## Format options

Dates are formatted with `Intl.DateTimeFormat` using the `locale` and `formatOptions` props.

<Example>
  <DatePicker label="Date" defaultValue={new Date()} formatOptions={{ dateStyle: 'full' }} />

  ```jsx
  <DatePicker label="Date" defaultValue={new Date()} formatOptions={{ dateStyle: 'full' }} />
  ```
</Example>

## Min and max

Dates outside `min` and `max` are disabled in the calendar.

<Example>
  <DatePicker
    label="Date"
    min={new Date(Date.now() - 7 * 86400000)}
    max={new Date(Date.now() + 7 * 86400000)}
  />

  ```jsx
  <DatePicker
    label="Date"
    min={new Date(Date.now() - 7 * 86400000)}
    max={new Date(Date.now() + 7 * 86400000)}
  />
  ```
</Example>

## Disabled

<Example>
  <DatePicker label="Date" disabled />

  ```jsx
  <DatePicker label="Date" disabled />
  ```
</Example>

## Controlled

<Example>
  <DatePickerExample />

  ```jsx
  const [date, setDate] = useState();

  <DatePicker label="Date" value={date} onValueChange={setDate} />
  ```
</Example>

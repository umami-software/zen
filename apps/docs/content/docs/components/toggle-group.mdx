import { Example, Lucide, ToggleGroup, ToggleGroupItem } from '../../../mdx-components';

A set of two-state buttons that can be toggled on or off. By default one item can be pressed at a
time; the group's value is always an array of the pressed items' `value`s.

<Example>
  <ToggleGroup variant="outline" defaultValue={['center']}>
    <ToggleGroupItem value="left" aria-label="Align left">
      <Lucide.AlignLeft />
    </ToggleGroupItem>
    <ToggleGroupItem value="center" aria-label="Align center">
      <Lucide.AlignCenter />
    </ToggleGroupItem>
    <ToggleGroupItem value="right" aria-label="Align right">
      <Lucide.AlignRight />
    </ToggleGroupItem>
  </ToggleGroup>

  ```jsx
  <ToggleGroup variant="outline" defaultValue={['center']}>
    <ToggleGroupItem value="left" aria-label="Align left">
      <Lucide.AlignLeft />
    </ToggleGroupItem>
    <ToggleGroupItem value="center" aria-label="Align center">
      <Lucide.AlignCenter />
    </ToggleGroupItem>
    <ToggleGroupItem value="right" aria-label="Align right">
      <Lucide.AlignRight />
    </ToggleGroupItem>
  </ToggleGroup>
  ```
</Example>

## Multiple

Pass `multiple` to let several items be pressed at once.

<Example>
  <ToggleGroup multiple variant="outline" defaultValue={['bold']}>
    <ToggleGroupItem value="bold" aria-label="Toggle bold">
      <Lucide.Bold />
    </ToggleGroupItem>
    <ToggleGroupItem value="italic" aria-label="Toggle italic">
      <Lucide.Italic />
    </ToggleGroupItem>
    <ToggleGroupItem value="underline" aria-label="Toggle underline">
      <Lucide.Underline />
    </ToggleGroupItem>
  </ToggleGroup>

  ```jsx
  <ToggleGroup multiple variant="outline" defaultValue={['bold']}>
    <ToggleGroupItem value="bold" aria-label="Toggle bold">
      <Lucide.Bold />
    </ToggleGroupItem>
    <ToggleGroupItem value="italic" aria-label="Toggle italic">
      <Lucide.Italic />
    </ToggleGroupItem>
    <ToggleGroupItem value="underline" aria-label="Toggle underline">
      <Lucide.Underline />
    </ToggleGroupItem>
  </ToggleGroup>
  ```
</Example>

## Spacing

`spacing` sets the gap between items. `spacing={0}` joins them into one segmented control.

<Example>
  <ToggleGroup variant="outline" spacing={0} defaultValue={['week']}>
    <ToggleGroupItem value="day">Day</ToggleGroupItem>
    <ToggleGroupItem value="week">Week</ToggleGroupItem>
    <ToggleGroupItem value="month">Month</ToggleGroupItem>
    <ToggleGroupItem value="year">Year</ToggleGroupItem>
  </ToggleGroup>

  ```jsx
  <ToggleGroup variant="outline" spacing={0} defaultValue={['week']}>
    <ToggleGroupItem value="day">Day</ToggleGroupItem>
    <ToggleGroupItem value="week">Week</ToggleGroupItem>
    <ToggleGroupItem value="month">Month</ToggleGroupItem>
    <ToggleGroupItem value="year">Year</ToggleGroupItem>
  </ToggleGroup>
  ```
</Example>

## Sizes

`variant` and `size` on the group apply to every item.

<Example>
  <ToggleGroup size="sm" defaultValue={['top']}>
    <ToggleGroupItem value="top">Top</ToggleGroupItem>
    <ToggleGroupItem value="bottom">Bottom</ToggleGroupItem>
  </ToggleGroup>
  <ToggleGroup defaultValue={['top']}>
    <ToggleGroupItem value="top">Top</ToggleGroupItem>
    <ToggleGroupItem value="bottom">Bottom</ToggleGroupItem>
  </ToggleGroup>
  <ToggleGroup size="lg" defaultValue={['top']}>
    <ToggleGroupItem value="top">Top</ToggleGroupItem>
    <ToggleGroupItem value="bottom">Bottom</ToggleGroupItem>
  </ToggleGroup>

  ```jsx
  <ToggleGroup size="sm" defaultValue={['top']}>
    <ToggleGroupItem value="top">Top</ToggleGroupItem>
    <ToggleGroupItem value="bottom">Bottom</ToggleGroupItem>
  </ToggleGroup>
  <ToggleGroup defaultValue={['top']}>
    <ToggleGroupItem value="top">Top</ToggleGroupItem>
    <ToggleGroupItem value="bottom">Bottom</ToggleGroupItem>
  </ToggleGroup>
  <ToggleGroup size="lg" defaultValue={['top']}>
    <ToggleGroupItem value="top">Top</ToggleGroupItem>
    <ToggleGroupItem value="bottom">Bottom</ToggleGroupItem>
  </ToggleGroup>
  ```
</Example>

## Controlled

```jsx
const [value, setValue] = useState(['left']);

<ToggleGroup value={value} onValueChange={setValue}>
  ...
</ToggleGroup>
```

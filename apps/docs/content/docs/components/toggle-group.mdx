A set of two-state buttons that can be toggled on or off.

<Example>
  <ToggleGroup selectionMode="single">
    <ToggleGroupItem id="one">One</ToggleGroupItem>
    <ToggleGroupItem id="two">Two</ToggleGroupItem>
    <ToggleGroupItem id="three">Three</ToggleGroupItem>
  </ToggleGroup>

  ```jsx
  <ToggleGroup selectionMode="single">
    <ToggleGroupItem id="one">One</ToggleGroupItem>
    <ToggleGroupItem id="two">Two</ToggleGroupItem>
    <ToggleGroupItem id="three">Three</ToggleGroupItem>
  </ToggleGroup>
  ```
</Example>

## Component props

### ToggleGroup

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the group</ResponseField>
  <ResponseField name="selectionMode" type="SelectionMode">"single" | "multiple" (default: "single")</ResponseField>
  <ResponseField name="value" type="string[]">Controlled selected values</ResponseField>
  <ResponseField name="defaultValue" type="string[]">Default selected values</ResponseField>
  <ResponseField name="variant" type="'default' | 'outline' | 'primary'">Item style, shared with items through context ("primary" is kept for backward compatibility)</ResponseField>
  <ResponseField name="size" type="'sm' | 'md' | 'lg'">Item size, shared with items through context</ResponseField>
  <ResponseField name="spacing" type="number">Gap between items in spacing units. `0` (default) renders a joined group.</ResponseField>
  <ResponseField name="orientation" type="'horizontal' | 'vertical'">Layout direction (default: "horizontal")</ResponseField>
  <ResponseField name="onChange" type="(value: string[]) => void">Called when selection changes</ResponseField>
</PropertiesTable>

### ToggleGroupItem

<PropertiesTable>
  <ResponseField name="id" type="string">Unique item identifier (required unless `value` is set)</ResponseField>
  <ResponseField name="variant" type="'default' | 'outline' | 'primary'">Overridden by the parent group when set there</ResponseField>
  <ResponseField name="size" type="'sm' | 'md' | 'lg'">Overridden by the parent group when set there</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the item</ResponseField>
</PropertiesTable>

import { Example, Toggle } from '../../../mdx-components';

A toggle button allows a user to toggle a selection on or off, for example switching between two states or modes.

<Example>
  <Toggle>B</Toggle>

  ```jsx
  <Toggle>B</Toggle>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the toggle</ResponseField>
  <ResponseField name="isSelected" type="boolean">Controlled selected state</ResponseField>
  <ResponseField name="defaultSelected" type="boolean">Default selected state</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the toggle</ResponseField>
  <ResponseField name="onChange" type="(isSelected: boolean) => void">Called when state changes</ResponseField>
</PropertiesTable>

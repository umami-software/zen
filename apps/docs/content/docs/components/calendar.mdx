import { Calendar, CalendarExample, Example } from '../../../mdx-components';

A calendar displays one or more date grids and allows users to select a single date.

<Example>
  <CalendarExample />

  ```jsx
  <Calendar onChange={(date) => console.log(date)} />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="value" type="Date">Controlled selected date</ResponseField>
  <ResponseField name="defaultValue" type="Date">Default selected date</ResponseField>
  <ResponseField name="minValue" type="Date">Minimum selectable date</ResponseField>
  <ResponseField name="maxValue" type="Date">Maximum selectable date</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the calendar</ResponseField>
  <ResponseField name="isReadOnly" type="boolean">Makes the calendar read-only</ResponseField>
  <ResponseField name="onChange" type="(date: Date) => void">Called when date changes</ResponseField>
</PropertiesTable>

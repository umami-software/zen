import { Button, Example, ToastActionExample, ToastExample, ToastTitleExample, ToastVariantsExample } from '../../../mdx-components';

A succinct message that is displayed temporarily.

<Example>
  <ToastExample />

  ```jsx
  <Button onPress={() => toast('Your message has been sent.')}>Show toast</Button>
  ```
</Example>

## With title

<Example>
  <ToastTitleExample />

  ```jsx
  <Button
    onPress={() =>
      toast('The application encountered an error.', { title: 'Oops! Something went wrong.' })
    }
  >
    Show toast
  </Button>
  ```
</Example>

## With action

<Example>
  <ToastActionExample />

  ```jsx
  <Button
    onPress={() =>
      toast('The task was deleted successfully.', {
        title: 'Task deleted',
        actions: ['Undo'],
      })
    }
  >
    Show toast
  </Button>
  ```
</Example>

## Variants

<Example>
  <ToastVariantsExample />

  ```jsx
  <>
    <Button onPress={() => toast('Your message was sent.', { variant: 'success' })}>
      Success toast
    </Button>
    <Button onPress={() => toast('Something went wrong.', { variant: 'error' })}>
      Error toast
    </Button>
  </>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="message" type="ReactNode">Toast message content</ResponseField>
  <ResponseField name="title" type="ReactNode">Optional toast title</ResponseField>
  <ResponseField name="variant" type="ToastVariant">"default" | "success" | "error"</ResponseField>
  <ResponseField name="action" type="ReactNode">Action element (e.g., button)</ResponseField>
  <ResponseField name="onAction" type="() => void">Called when action is clicked</ResponseField>
</PropertiesTable>

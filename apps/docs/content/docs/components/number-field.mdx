---
title: Number Field
---

import { Box, Example, NumberField } from '../../../mdx-components';

A number field allows a user to enter a numeric value with stepper buttons for incrementing and decrementing.

<Example>
  <Box width="200px"><NumberField label="Quantity" defaultValue={1} /></Box>

  ```jsx
  <NumberField label="Quantity" defaultValue={1} />
  ```
</Example>

## Min, max, and step

<Example>
  <Box width="200px"><NumberField label="Percent" defaultValue={50} minValue={0} maxValue={100} step={10} /></Box>

  ```jsx
  <NumberField label="Percent" defaultValue={50} minValue={0} maxValue={100} step={10} />
  ```
</Example>

## Format options

Values are formatted with `Intl.NumberFormat` using the `formatOptions` prop.

<Example>
  <Box width="200px"><NumberField label="Price" defaultValue={25} formatOptions={{ style: 'currency', currency: 'USD' }} /></Box>

  ```jsx
  <NumberField
    label="Price"
    defaultValue={25}
    formatOptions={{ style: 'currency', currency: 'USD' }}
  />
  ```
</Example>

## Disabled

<Example>
  <Box width="200px"><NumberField label="Quantity" defaultValue={1} isDisabled /></Box>

  ```jsx
  <NumberField label="Quantity" defaultValue={1} isDisabled />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the field</ResponseField>
  <ResponseField name="placeholder" type="string">Placeholder text</ResponseField>
  <ResponseField name="value" type="number">Controlled value</ResponseField>
  <ResponseField name="defaultValue" type="number">Default value</ResponseField>
  <ResponseField name="minValue" type="number">Minimum value</ResponseField>
  <ResponseField name="maxValue" type="number">Maximum value</ResponseField>
  <ResponseField name="step" type="number">Amount to increment or decrement</ResponseField>
  <ResponseField name="formatOptions" type="Intl.NumberFormatOptions">Number format options</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the field</ResponseField>
  <ResponseField name="isReadOnly" type="boolean">Prevents changing the value</ResponseField>
  <ResponseField name="isRequired" type="boolean">Marks the field as required</ResponseField>
  <ResponseField name="onChange" type="(value: number | null) => void">Called when the value changes</ResponseField>
</PropertiesTable>

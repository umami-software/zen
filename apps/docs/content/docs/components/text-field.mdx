## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the field</ResponseField>
  <ResponseField name="description" type="ReactNode">Helper text rendered below the field and linked with `aria-describedby`</ResponseField>
  <ResponseField name="error" type="ReactNode">Error message rendered below the field. Sets `aria-invalid` and `role="alert"`</ResponseField>
  <ResponseField name="placeholder" type="string">Placeholder text</ResponseField>
  <ResponseField name="value" type="string">Controlled value</ResponseField>
  <ResponseField name="defaultValue" type="string">Default value (uncontrolled)</ResponseField>
  <ResponseField name="variant" type="TextFieldVariant">"quiet" | "none"</ResponseField>
  <ResponseField name="asTextArea" type="boolean">Renders as a textarea instead of input</ResponseField>
  <ResponseField name="resize" type="ResizeDirection">"vertical" | "horizontal" | "both" | "none"</ResponseField>
  <ResponseField name="allowCopy" type="boolean">Shows a copy button</ResponseField>
  <ResponseField name="isReadOnly" type="boolean">Makes the field read-only</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the field</ResponseField>
  <ResponseField name="onChange" type="(value: string) => void">Called when value changes</ResponseField>
</PropertiesTable>

## Controlled

`onValueChange` receives the string value. `onChange` is the native change event.

```jsx
const [value, setValue] = useState('');

<TextField label="Name" value={value} onValueChange={setValue} />
```

## Component props

Native `input` / `textarea` props are passed to the control.

<PropertiesTable>
  <PropertiesTable.Row name="label" type="ReactNode">Label displayed above the field</PropertiesTable.Row>
  <PropertiesTable.Row name="description" type="ReactNode">Helper text rendered below the field and linked with `aria-describedby`</PropertiesTable.Row>
  <PropertiesTable.Row name="error" type="ReactNode">Error message rendered below the field. Sets `aria-invalid`</PropertiesTable.Row>
  <PropertiesTable.Row name="placeholder" type="string">Placeholder text</PropertiesTable.Row>
  <PropertiesTable.Row name="value" type="string">Controlled value</PropertiesTable.Row>
  <PropertiesTable.Row name="defaultValue" type="string">Initial value (uncontrolled)</PropertiesTable.Row>
  <PropertiesTable.Row name="onValueChange" type="(value: string) => void">Called with the new value</PropertiesTable.Row>
  <PropertiesTable.Row name="onChange" type="(event: ChangeEvent) => void">Native change event</PropertiesTable.Row>
  <PropertiesTable.Row name="variant" type="string">"default" | "quiet" (default: "default")</PropertiesTable.Row>
  <PropertiesTable.Row name="asTextArea" type="boolean">Renders a textarea instead of an input</PropertiesTable.Row>
  <PropertiesTable.Row name="resize" type="string">"vertical" | "horizontal" | "both" | "none"</PropertiesTable.Row>
  <PropertiesTable.Row name="allowCopy" type="boolean">Shows a copy button</PropertiesTable.Row>

  <PropertiesTable.Row name="disabled" type="boolean">Disables the field</PropertiesTable.Row>
</PropertiesTable>

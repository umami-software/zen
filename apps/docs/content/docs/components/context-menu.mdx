import { Box, ContextMenu, Example, Menu, MenuItem, MenuSeparator } from '../../../mdx-components';

A context menu opens a menu at the pointer position when a user right clicks the trigger area.
Pass two children: the trigger element followed by a `Menu`.

<Example>
  <ContextMenu>
    <Box
      border
      borderRadius="md"
      padding="8"
      style={{ borderStyle: 'dashed' }}
    >
      Right click here
    </Box>
    <Menu>
      <MenuItem>Cut</MenuItem>
      <MenuItem>Copy</MenuItem>
      <MenuItem>Paste</MenuItem>
      <MenuSeparator />
      <MenuItem>Delete</MenuItem>
    </Menu>
  </ContextMenu>

  ```jsx
  <ContextMenu>
    <Box border borderRadius="md" padding="8" style={{ borderStyle: 'dashed' }}>
      Right click here
    </Box>
    <Menu>
      <MenuItem>Cut</MenuItem>
      <MenuItem>Copy</MenuItem>
      <MenuItem>Paste</MenuItem>
      <MenuSeparator />
      <MenuItem>Delete</MenuItem>
    </Menu>
  </ContextMenu>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="children" type="ReactNode">Trigger element followed by a Menu</ResponseField>
  <ResponseField name="isOpen" type="boolean">Controlled open state</ResponseField>
  <ResponseField name="onOpenChange" type="(open: boolean) => void">Called when the menu opens or closes</ResponseField>
</PropertiesTable>

Remaining root props (`defaultOpen`, `actionsRef`, `orientation`, …) are forwarded to the underlying
Base UI `ContextMenu.Root`.

## Parts

<PropertiesTable>
  <ResponseField name="ContextMenuRoot" type="component">The Base UI root, for explicit composition</ResponseField>
  <ResponseField name="ContextMenuTrigger" type="component">The right-click target</ResponseField>
  <ResponseField name="ContextMenuContent" type="component">The menu popup. Accepts all `Menu` props</ResponseField>
</PropertiesTable>

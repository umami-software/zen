import { Box, Example, PasswordField } from '../../../mdx-components';

A secure text input for passwords with a visibility toggle button. Allows users to show or hide their password while typing, improving usability without compromising security.

<Example>
  <Box width="300px">
    <PasswordField />
  </Box>

  ```jsx
  <PasswordField />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the field</ResponseField>
  <ResponseField name="value" type="string">Controlled value</ResponseField>
  <ResponseField name="defaultValue" type="string">Default value (uncontrolled)</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the field</ResponseField>
  <ResponseField name="isReadOnly" type="boolean">Makes the field read-only</ResponseField>
  <ResponseField name="isRevealed" type="boolean">Controls whether the value is visible</ResponseField>
  <ResponseField name="defaultRevealed" type="boolean">Initial visibility (uncontrolled)</ResponseField>
  <ResponseField name="onRevealChange" type="(revealed: boolean) => void">Called when the visibility toggle is pressed</ResponseField>
  <ResponseField name="autoComplete" type="string">Defaults to "current-password"</ResponseField>
  <ResponseField name="onChange" type="(value: string) => void">Called when value changes</ResponseField>
</PropertiesTable>

import { Box, Example, PasswordField } from '../../../mdx-components';

A secure text input for passwords with a visibility toggle button. Allows users to show or hide their password while typing, improving usability without compromising security. It is built on the shadcn `InputGroup`.

<Example>
  <Box width="300px">
    <PasswordField />
  </Box>

  ```jsx
  <PasswordField />
  ```
</Example>

## Label

<Example>
  <Box width="300px">

  </Box>

  ```jsx
  <PasswordField label="Password" defaultValue="hunter2" />
  ```
</Example>

## Component props

Native `input` props are passed to the input. `onChange` receives the native change event; use
`onValueChange` for the string value.

<PropertiesTable>
  <PropertiesTable.Row name="label" type="ReactNode">Label displayed above the field</PropertiesTable.Row>
  <PropertiesTable.Row name="value" type="string">Controlled value</PropertiesTable.Row>
  <PropertiesTable.Row name="defaultValue" type="string">Initial value (uncontrolled)</PropertiesTable.Row>
  <PropertiesTable.Row name="onValueChange" type="(value: string) => void">Called with the new value</PropertiesTable.Row>
  <PropertiesTable.Row name="disabled" type="boolean">Disables the field</PropertiesTable.Row>
  <PropertiesTable.Row name="readOnly" type="boolean">Makes the field read-only</PropertiesTable.Row>
  <PropertiesTable.Row name="revealed" type="boolean">Controls whether the value is visible</PropertiesTable.Row>
  <PropertiesTable.Row name="defaultRevealed" type="boolean">Initial visibility (uncontrolled)</PropertiesTable.Row>
  <PropertiesTable.Row name="onRevealedChange" type="(revealed: boolean) => void">Called when the visibility toggle is pressed</PropertiesTable.Row>
  <PropertiesTable.Row name="autoComplete" type="string">Defaults to "current-password"</PropertiesTable.Row>
</PropertiesTable>

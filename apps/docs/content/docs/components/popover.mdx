import { Button, Example, Field, FieldGroup, FieldLabel, Input, Popover, PopoverContent, PopoverDescription, PopoverHeader, PopoverTitle, PopoverTrigger } from '../../../mdx-components';

Displays rich content in a portal, triggered by a button. Use the `render` prop on
`PopoverTrigger` to render it as your own button.

<Example>
  <Popover>
    <PopoverTrigger render={<Button variant="outline" />}>Open popover</PopoverTrigger>
    <PopoverContent>
      <PopoverHeader>
        <PopoverTitle>Dimensions</PopoverTitle>
        <PopoverDescription>Set the dimensions for the layer.</PopoverDescription>
      </PopoverHeader>
      <FieldGroup>
        <Field orientation="horizontal">
          <FieldLabel htmlFor="popover-width">Width</FieldLabel>
          <Input id="popover-width" defaultValue="100%" />
        </Field>
        <Field orientation="horizontal">
          <FieldLabel htmlFor="popover-height">Height</FieldLabel>
          <Input id="popover-height" defaultValue="25px" />
        </Field>
      </FieldGroup>
    </PopoverContent>
  </Popover>


## Position

`side` and `align` on `PopoverContent` position the popup relative to the trigger. `sideOffset`
and `alignOffset` add distance in pixels.

<Example direction="row">
  <Popover>
    <PopoverTrigger render={<Button variant="outline" />}>Top</PopoverTrigger>
    <PopoverContent side="top">
      <PopoverTitle>Top</PopoverTitle>
    </PopoverContent>
  </Popover>
  <Popover>
    <PopoverTrigger render={<Button variant="outline" />}>Right, start</PopoverTrigger>
    <PopoverContent side="right" align="start">
      <PopoverTitle>Right, start</PopoverTitle>
    </PopoverContent>
  </Popover>

  ```jsx
  <Popover>
    <PopoverTrigger render={<Button variant="outline" />}>Top</PopoverTrigger>
    <PopoverContent side="top">
      <PopoverTitle>Top</PopoverTitle>
    </PopoverContent>
  </Popover>
  <Popover>
    <PopoverTrigger render={<Button variant="outline" />}>Right, start</PopoverTrigger>
    <PopoverContent side="right" align="start">
      <PopoverTitle>Right, start</PopoverTitle>
    </PopoverContent>
  </Popover>
  ```
</Example>

## Controlled

Pass `open` and `onOpenChange` to `Popover` to control it.

```jsx
const [open, setOpen] = useState(false);

<Popover open={open} onOpenChange={setOpen}>
  <PopoverTrigger render={<Button variant="outline" />}>Open popover</PopoverTrigger>
  <PopoverContent>...</PopoverContent>
</Popover>
```

### PopoverTrigger

<PropertiesTable>
  <PropertiesTable.Row name="render" type="ReactElement | function">Replaces the rendered `button`, e.g. `render={<Button variant="outline" />}`</PropertiesTable.Row>
  <PropertiesTable.Row name="openOnHover" type="boolean">Also opens the popover on hover</PropertiesTable.Row>
</PropertiesTable>

### PopoverContent

<PropertiesTable>
  <PropertiesTable.Row name="side" type="string">"top" | "right" | "bottom" | "left" | "inline-start" | "inline-end" (default: "bottom")</PropertiesTable.Row>
  <PropertiesTable.Row name="align" type="string">"start" | "center" | "end" (default: "center")</PropertiesTable.Row>
  <PropertiesTable.Row name="sideOffset" type="number">Distance from the trigger in pixels (default: 4)</PropertiesTable.Row>
  <PropertiesTable.Row name="alignOffset" type="number">Offset along the alignment axis in pixels (default: 0)</PropertiesTable.Row>
</PropertiesTable>

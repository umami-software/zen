import { Button, Column, Dialog, DialogTrigger, Example, Popover, Switch } from '../../../mdx-components';

A popover is an overlay element positioned relative to a trigger.

<Example>
  <DialogTrigger overlayType="popover">
    <Button variant="outline">Settings</Button>
    <Popover>
      <Dialog>
        <Column gap="3">
          <Switch defaultSelected>Wi-Fi</Switch>
          <Switch defaultSelected>Bluetooth</Switch>
          <Switch>Mute</Switch>
        </Column>
      </Dialog>
    </Popover>
  </DialogTrigger>

  ```jsx
  <DialogTrigger overlayType="popover">

  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="placement" type="Placement">"top" | "bottom" | "left" | "right" | "start" | "end"</ResponseField>
  <ResponseField name="offset" type="number">Distance from the trigger element</ResponseField>
  <ResponseField name="isFullscreen" type="boolean">Opens popover in fullscreen mode</ResponseField>
</PropertiesTable>

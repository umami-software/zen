import { Box, Example, ListItem, Select, SelectSearchExample, SelectValueExample } from '../../../mdx-components';

Displays a list of options for the user to pick from—triggered by a button.

<Example>
  <Box width="300px">
    <Select placeholder="Please select an option">
      <ListItem id="one">One</ListItem>
      <ListItem id="two">Two</ListItem>
      <ListItem id="three">Three</ListItem>
      <ListItem id="four">Four</ListItem>
      <ListItem id="five">Five</ListItem>
    </Select>
  </Box>

  ```jsx
  <Select placeholder="Please select an option">
    <ListItem id="one">One</ListItem>
    <ListItem id="two">Two</ListItem>
    <ListItem id="three">Three</ListItem>
  </Select>
  ```
</Example>

## Selected item alignment

By default, the menu opens below the trigger like a dropdown. Set `alignItemWithTrigger` to align
the selected option with the trigger value instead.

<Example>
  <Box width="300px">
    <Select defaultValue="three" alignItemWithTrigger>
      <ListItem id="one">One</ListItem>
      <ListItem id="two">Two</ListItem>
      <ListItem id="three">Three</ListItem>
      <ListItem id="four">Four</ListItem>
      <ListItem id="five">Five</ListItem>
    </Select>
  </Box>

  ```jsx
  <Select defaultValue="three" alignItemWithTrigger>
    <ListItem id="one">One</ListItem>
    <ListItem id="two">Two</ListItem>
    <ListItem id="three">Three</ListItem>
    <ListItem id="four">Four</ListItem>
    <ListItem id="five">Five</ListItem>
  </Select>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="value" type="string | number">Controlled selected value</ResponseField>
  <ResponseField name="defaultValue" type="string | number">Default selected value</ResponseField>
  <ResponseField name="label" type="string">Label text displayed above the select</ResponseField>
  <ResponseField name="placeholder" type="string">Placeholder text when no selection</ResponseField>
  <ResponseField name="isLoading" type="boolean">Shows loading indicator</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the select</ResponseField>
  <ResponseField name="isFullscreen" type="boolean">Opens options in fullscreen mode</ResponseField>
  <ResponseField name="allowSearch" type="boolean">Enables search within options</ResponseField>
  <ResponseField name="searchValue" type="string">Controlled search field value</ResponseField>
  <ResponseField name="searchDelay" type="number">Debounce delay for search</ResponseField>
  <ResponseField name="maxHeight" type="string | number">Max height of the dropdown list</ResponseField>
  <ResponseField name="showIcon" type="boolean">Show chevron icon on trigger button (default: true)</ResponseField>
  <ResponseField name="alignItemWithTrigger" type="boolean">Aligns the selected option with the trigger instead of opening below it (default: false)</ResponseField>
  <ResponseField name="onChange" type="(value: any) => void">Called when selection changes</ResponseField>
  <ResponseField name="onSearch" type="(value: string) => void">Called when search value changes</ResponseField>
  <ResponseField name="renderValue" type="ReactNode | Function">Custom render for selected value</ResponseField>
  <ResponseField name="buttonProps" type="ButtonProps">Props passed to the trigger button</ResponseField>
  <ResponseField name="listProps" type="ListProps">Props passed to the list component</ResponseField>
  <ResponseField name="popoverProps" type="PopoverProps">Props passed to the popover</ResponseField>
</PropertiesTable>

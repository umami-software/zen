import { Box, Example, Select, SelectContent, SelectGroup, SelectItem, SelectLabel, SelectSeparator, SelectTrigger, SelectValue, SelectValueExample } from '../../../mdx-components';

Displays a list of options for the user to pick from, triggered by a button. Pass `items` to
`Select`, a map of value to label, so `SelectValue` can show the label of the selected value.

<Example>
  <Box width="192px">
    <Select items={{ apple: 'Apple', banana: 'Banana', blueberry: 'Blueberry', grapes: 'Grapes', pineapple: 'Pineapple' }}>
      <SelectTrigger className="w-full">
        <SelectValue placeholder="Select a fruit" />
      </SelectTrigger>
      <SelectContent>
        <SelectItem value="apple">Apple</SelectItem>
        <SelectItem value="banana">Banana</SelectItem>
        <SelectItem value="blueberry">Blueberry</SelectItem>
        <SelectItem value="grapes">Grapes</SelectItem>
        <SelectItem value="pineapple">Pineapple</SelectItem>
      </SelectContent>
    </Select>
  </Box>

  ```jsx
  <Select items={{ apple: 'Apple', banana: 'Banana', blueberry: 'Blueberry', grapes: 'Grapes', pineapple: 'Pineapple' }}>
    <SelectTrigger className="w-full">
      <SelectValue placeholder="Select a fruit" />
    </SelectTrigger>
    <SelectContent>
      <SelectItem value="apple">Apple</SelectItem>
      <SelectItem value="banana">Banana</SelectItem>
      <SelectItem value="blueberry">Blueberry</SelectItem>
      <SelectItem value="grapes">Grapes</SelectItem>
      <SelectItem value="pineapple">Pineapple</SelectItem>
    </SelectContent>
  </Select>
  ```
</Example>

          <SelectItem value="carrot">Carrot</SelectItem>

        </SelectGroup>

    </Select>

## Size and disabled

`SelectTrigger` takes `size="sm"` or `size="default"`. Disable the whole select with `disabled`
on `Select`, or a single option with `disabled` on its `SelectItem`.

<Example direction="row">

    <SelectTrigger size="sm">
      <SelectValue />
    </SelectTrigger>

      <SelectItem value="light">Light</SelectItem>
      <SelectItem value="dark">Dark</SelectItem>
      <SelectItem value="system" disabled>
        System
      </SelectItem>

  </Select>

## Popup alignment

By default the popup aligns the selected item with the trigger, like a native select. Pass
`alignItemWithTrigger={false}` to `SelectContent` to open it below the trigger like a dropdown,
positioned with `side` and `align`.

<Example>
  <Box width="192px">
    <Select defaultValue="three" items={{ one: 'One', two: 'Two', three: 'Three', four: 'Four', five: 'Five' }}>
      <SelectTrigger className="w-full">
        <SelectValue />
      </SelectTrigger>
      <SelectContent alignItemWithTrigger={false} align="start">
        <SelectItem value="one">One</SelectItem>
        <SelectItem value="two">Two</SelectItem>
        <SelectItem value="three">Three</SelectItem>
        <SelectItem value="four">Four</SelectItem>
        <SelectItem value="five">Five</SelectItem>
      </SelectContent>
    </Select>
  </Box>

  ```jsx
  <Select defaultValue="three" items={{ one: 'One', two: 'Two', three: 'Three', four: 'Four', five: 'Five' }}>
    <SelectTrigger className="w-full">
      <SelectValue />
    </SelectTrigger>
    <SelectContent alignItemWithTrigger={false} align="start">
      <SelectItem value="one">One</SelectItem>
      <SelectItem value="two">Two</SelectItem>
      <SelectItem value="three">Three</SelectItem>
      <SelectItem value="four">Four</SelectItem>
      <SelectItem value="five">Five</SelectItem>
    </SelectContent>
  </Select>
  ```
</Example>

## Custom value

Pass a function as the child of `SelectValue` to render the selected value yourself.

<Example>
  <SelectValueExample />

  ```jsx
  const statuses = [
    { value: 'online', label: 'Online', variant: 'success' },
    { value: 'away', label: 'Away', variant: 'warning' },
    { value: 'busy', label: 'Busy', variant: 'destructive' },
    { value: 'offline', label: 'Offline', variant: 'muted' },
  ];

  function StatusLabel({ value }) {
    const status = statuses.find(item => item.value === value);

    return status ? <StatusLight variant={status.variant}>{status.label}</StatusLight> : null;
  }

  <Select defaultValue="online">
    <SelectTrigger className="w-full">
      <SelectValue>{value => <StatusLabel value={value} />}</SelectValue>
    </SelectTrigger>
    <SelectContent>
      {statuses.map(status => (
        <SelectItem key={status.value} value={status.value}>
          <StatusLabel value={status.value} />
        </SelectItem>
      ))}
    </SelectContent>
  </Select>
  ```
</Example>

## Controlled

```jsx
const [value, setValue] = useState(null);

<Select value={value} onValueChange={setValue} items={items}>
  ...
</Select>
```

To filter the options as the user types, use [Combobox](/docs/components/combobox) instead.

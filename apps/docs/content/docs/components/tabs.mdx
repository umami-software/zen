import { Box, Example, Lucide, Tabs, TabsContent, TabsList, TabsTrigger, Text } from '../../../mdx-components';

A set of layered sections of content, known as tab panels, that are displayed one at a time. Each
`TabsTrigger` is paired with the `TabsContent` that has the same `value`.

<Example>
  <Box width="400px" maxWidth="100%">
    <Tabs defaultValue="overview">
      <TabsList>
        <TabsTrigger value="overview">Overview</TabsTrigger>
        <TabsTrigger value="analytics">Analytics</TabsTrigger>
        <TabsTrigger value="reports">Reports</TabsTrigger>
      </TabsList>
      <TabsContent value="overview">
        <Text size="sm" color="muted-foreground">View your key metrics and recent project activity.</Text>
      </TabsContent>
      <TabsContent value="analytics">
        <Text size="sm" color="muted-foreground">Track performance and user engagement.</Text>
      </TabsContent>
      <TabsContent value="reports">
        <Text size="sm" color="muted-foreground">Generate and download your detailed reports.</Text>
      </TabsContent>
    </Tabs>
  </Box>

  ```jsx
  <Tabs defaultValue="overview">
    <TabsList>
      <TabsTrigger value="overview">Overview</TabsTrigger>
      <TabsTrigger value="analytics">Analytics</TabsTrigger>
      <TabsTrigger value="reports">Reports</TabsTrigger>
    </TabsList>
    <TabsContent value="overview">
      <Text size="sm" color="muted-foreground">View your key metrics and recent project activity.</Text>
    </TabsContent>
    <TabsContent value="analytics">
      <Text size="sm" color="muted-foreground">Track performance and user engagement.</Text>
    </TabsContent>
    <TabsContent value="reports">
      <Text size="sm" color="muted-foreground">Generate and download your detailed reports.</Text>
    </TabsContent>
  </Tabs>
  ```
</Example>

## Line

Use `variant="line"` on `TabsList` for underlined tabs without a background.

<Example>
  <Tabs defaultValue="overview">
    <TabsList variant="line">
      <TabsTrigger value="overview">Overview</TabsTrigger>
      <TabsTrigger value="analytics">Analytics</TabsTrigger>
      <TabsTrigger value="reports">Reports</TabsTrigger>
    </TabsList>
  </Tabs>

  ```jsx
  <Tabs defaultValue="overview">
    <TabsList variant="line">
      <TabsTrigger value="overview">Overview</TabsTrigger>
      <TabsTrigger value="analytics">Analytics</TabsTrigger>
      <TabsTrigger value="reports">Reports</TabsTrigger>
    </TabsList>
  </Tabs>
  ```
</Example>

## Controlled

```jsx
const [tab, setTab] = useState('overview');

<Tabs value={tab} onValueChange={setTab}>
  ...
</Tabs>
```

## Component props

### Tabs

<PropertiesTable>
  <PropertiesTable.Row name="value" type="any">Controlled active tab</PropertiesTable.Row>
  <PropertiesTable.Row name="defaultValue" type="any">Initial active tab when uncontrolled (default: 0)</PropertiesTable.Row>
  <PropertiesTable.Row name="onValueChange" type="(value, details) => void">Called when the active tab changes</PropertiesTable.Row>
  <PropertiesTable.Row name="orientation" type="string">"horizontal" | "vertical" (default: "horizontal")</PropertiesTable.Row>
</PropertiesTable>

### TabsList

<PropertiesTable>
  <PropertiesTable.Row name="variant" type="string">"default" | "line" (default: "default")</PropertiesTable.Row>
  <PropertiesTable.Row name="activateOnFocus" type="boolean">Activates a tab when it receives keyboard focus (default: false)</PropertiesTable.Row>
</PropertiesTable>

### TabsTrigger

<PropertiesTable>
  <PropertiesTable.Row name="value" type="any">Matches the `value` of a `TabsContent` (required)</PropertiesTable.Row>
  <PropertiesTable.Row name="disabled" type="boolean">Disables the tab</PropertiesTable.Row>
</PropertiesTable>

### TabsContent

<PropertiesTable>
  <PropertiesTable.Row name="value" type="any">Matches the `value` of a `TabsTrigger` (required)</PropertiesTable.Row>
  <PropertiesTable.Row name="keepMounted" type="boolean">Keeps the panel in the DOM while hidden (default: false)</PropertiesTable.Row>
</PropertiesTable>

import { Example, Tab, TabList, TabPanel, Tabs } from '../../../mdx-components';

Tabs organize content into multiple sections and allow users to navigate between them.

<Example>
  <Tabs>
    <TabList>
      <Tab id="one">One</Tab>
      <Tab id="two">Two</Tab>
      <Tab id="three">Three</Tab>
    </TabList>
    <TabPanel id="one">Content for tab one.</TabPanel>
    <TabPanel id="two">Content for tab two.</TabPanel>
    <TabPanel id="three">Content for tab three.</TabPanel>
  </Tabs>

  ```jsx
  <Tabs>
    <TabList>
      <Tab id="one">One</Tab>
      <Tab id="two">Two</Tab>
      <Tab id="three">Three</Tab>
    </TabList>
    <TabPanel id="one">Content for tab one.</TabPanel>
    <TabPanel id="two">Content for tab two.</TabPanel>
    <TabPanel id="three">Content for tab three.</TabPanel>
  </Tabs>
  ```
</Example>

## Component props

### Tabs

<PropertiesTable>
  <ResponseField name="selectedKey" type="string">Controlled selected tab key</ResponseField>
  <ResponseField name="defaultSelectedKey" type="string">Default selected tab key</ResponseField>
  <ResponseField name="onSelectionChange" type="(key: string) => void">Called when selection changes</ResponseField>
</PropertiesTable>

### Tab

<PropertiesTable>
  <ResponseField name="id" type="string">Unique tab identifier (required)</ResponseField>
  <ResponseField name="href" type="string">Makes the tab a link</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the tab</ResponseField>
</PropertiesTable>

import { Example, Menu, MenuItem, Navbar, NavbarItem } from '../../../mdx-components';

A horizontal navigation bar for website headers. Built on an accessible navigation menu primitive, it supports dropdown menus, links, and nested navigation items with a shared animated dropdown, full keyboard navigation, and hover/focus interactions.

<Example>
  <Navbar>
    <NavbarItem label="Product">
      <Menu>
        <MenuItem>Features</MenuItem>
        <MenuItem>Pricing</MenuItem>
      </Menu>
    </NavbarItem>
    <NavbarItem label="Resources">
      <Menu>
        <MenuItem>Documentation</MenuItem>
        <MenuItem>API Reference</MenuItem>
        <MenuItem>Support</MenuItem>
      </Menu>
    </NavbarItem>
    <NavbarItem label="Company">
      <Menu>
        <MenuItem>About</MenuItem>
        <MenuItem>Blog</MenuItem>
        <MenuItem>Careers</MenuItem>
      </Menu>
    </NavbarItem>
    <NavbarItem>
      <a href="https://github.com/umami-software/umami">GitHub</a>
    </NavbarItem>

  </Navbar>

## Component props

### Navbar

<PropertiesTable>
  <ResponseField name="showArrow" type="boolean">Shows dropdown arrow on items (default: true)</ResponseField>
</PropertiesTable>

### NavbarItem

<PropertiesTable>
  <ResponseField name="label" type="string">Item label. When set, the item becomes a trigger and its children render in the dropdown.</ResponseField>
</PropertiesTable>

### NavbarLink

A semantic navigation link that renders an `<a>` element and participates in keyboard navigation. Use it for direct links instead of a plain anchor.

<PropertiesTable>
  <ResponseField name="href" type="string">The destination URL.</ResponseField>
  <ResponseField name="active" type="boolean">Marks the link as the current page (default: false).</ResponseField>
</PropertiesTable>

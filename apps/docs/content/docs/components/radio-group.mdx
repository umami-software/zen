import { Example, Field, FieldContent, FieldDescription, FieldLabel, RadioGroup, RadioGroupItem } from '../../../mdx-components';

A set of checkable buttons, known as radio buttons, where no more than one can be checked at a
time. `RadioGroupItem` renders only the radio, so pair each one with a `FieldLabel`.

<Example>
  <RadioGroup defaultValue="comfortable" className="w-fit">
    <Field orientation="horizontal">
      <RadioGroupItem value="default" id="r1" />
      <FieldLabel htmlFor="r1">Default</FieldLabel>
    </Field>
    <Field orientation="horizontal">
      <RadioGroupItem value="comfortable" id="r2" />
      <FieldLabel htmlFor="r2">Comfortable</FieldLabel>
    </Field>
    <Field orientation="horizontal">
      <RadioGroupItem value="compact" id="r3" />
      <FieldLabel htmlFor="r3">Compact</FieldLabel>
    </Field>
  </RadioGroup>

  ```jsx
  <RadioGroup defaultValue="comfortable" className="w-fit">
  ```

</Example>

## With description

Wrap the label and a `FieldDescription` in `FieldContent`.

<Example>
  <RadioGroup defaultValue="monthly" className="w-fit">
    <Field orientation="horizontal">
      <RadioGroupItem value="monthly" id="plan-monthly" />
      <FieldContent>
        <FieldLabel htmlFor="plan-monthly">Monthly</FieldLabel>
        <FieldDescription>Billed every month. Cancel any time.</FieldDescription>
      </FieldContent>
    </Field>
    <Field orientation="horizontal">
      <RadioGroupItem value="yearly" id="plan-yearly" />
      <FieldContent>
        <FieldLabel htmlFor="plan-yearly">Yearly</FieldLabel>
        <FieldDescription>Billed once a year. Two months free.</FieldDescription>
      </FieldContent>
    </Field>
  </RadioGroup>

  ```jsx

  ```
</Example>

## Disabled

Disable the whole group with `disabled` on `RadioGroup`, or a single option on its `RadioGroupItem`.

<Example>
  <RadioGroup defaultValue="email" className="w-fit">
    <Field orientation="horizontal">
      <RadioGroupItem value="email" id="notify-email" />
      <FieldLabel htmlFor="notify-email">Email</FieldLabel>
    </Field>
    <Field orientation="horizontal" data-disabled="true">
      <RadioGroupItem value="sms" id="notify-sms" disabled />
      <FieldLabel htmlFor="notify-sms">SMS</FieldLabel>
    </Field>
  </RadioGroup>

  ```jsx
  <RadioGroup defaultValue="email" className="w-fit">
    <Field orientation="horizontal">
      <RadioGroupItem value="email" id="notify-email" />
      <FieldLabel htmlFor="notify-email">Email</FieldLabel>
    </Field>
    <Field orientation="horizontal" data-disabled="true">
      <RadioGroupItem value="sms" id="notify-sms" disabled />
      <FieldLabel htmlFor="notify-sms">SMS</FieldLabel>
    </Field>
  </RadioGroup>
  ```
</Example>

## Controlled

```jsx
const [value, setValue] = useState('comfortable');

<RadioGroup value={value} onValueChange={setValue}>
  ...
</RadioGroup>
```

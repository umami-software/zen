title: Radio Group

import { Example, Radio, RadioGroup } from '../../../mdx-components';

A radio group allows a user to select a single item from a list of mutually exclusive options.

<Example>
  <RadioGroup>
    <Radio value="one">One</Radio>
    <Radio value="two">Two</Radio>
    <Radio value="three">Three</Radio>
  </RadioGroup>

  ```jsx
  <RadioGroup>
    <Radio value="one">One</Radio>
    <Radio value="two">Two</Radio>
    <Radio value="three">Three</Radio>
  </RadioGroup>
  ```
</Example>

## Orientation

<Example>
  <RadioGroup orientation="horizontal">
    <Radio value="one">One</Radio>
    <Radio value="two">Two</Radio>
    <Radio value="three">Three</Radio>
  </RadioGroup>

  ```jsx
  <RadioGroup orientation="horizontal">
    <Radio value="one">One</Radio>
    <Radio value="two">Two</Radio>
    <Radio value="three">Three</Radio>
  </RadioGroup>
  ```
</Example>

## Disabled

<Example>
  <RadioGroup>
    <Radio value="one">One</Radio>
    <Radio value="two">Two</Radio>
    <Radio value="three" isDisabled>Three</Radio>

  ```jsx
  <RadioGroup>
  ```

## Component props

### RadioGroup

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the group</ResponseField>
  <ResponseField name="value" type="string">Controlled selected value</ResponseField>
  <ResponseField name="defaultValue" type="string">Default selected value</ResponseField>
  <ResponseField name="orientation" type="Orientation">"horizontal" | "vertical"</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables all radios in the group</ResponseField>
  <ResponseField name="onChange" type="(value: string) => void">Called when selection changes</ResponseField>
</PropertiesTable>

### Radio

<PropertiesTable>
  <ResponseField name="value" type="string">Value of this radio option (required)</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables this radio option</ResponseField>
</PropertiesTable>

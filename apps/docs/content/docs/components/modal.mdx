import { Button, DataColumn, DataTable, Dialog, DialogTrigger, Example, Flexbox, Heading, Modal, ZenProvider } from '../../../mdx-components';

<ZenProvider>

A modal is an overlay element which blocks interaction with elements outside it.
Use `Sheet` for panels that slide in from an edge of the viewport.

<Example>
  <DialogTrigger>
    <Button>Center</Button>
    <Modal>
      <Dialog>
        <Flexbox justifyContent="center" alignItems="center" paddingX="9" paddingY="8">
          <Heading>Hello.</Heading>
        </Flexbox>
      </Dialog>
    </Modal>
  </DialogTrigger>
  <DialogTrigger>
    <Button>Fullscreen</Button>
    <Modal placement="fullscreen">
      <Dialog className="size-full rounded-none border-0 shadow-none">
        <Flexbox justifyContent="center" alignItems="center" paddingX="9" paddingY="8">
          <Heading>Hello.</Heading>
        </Flexbox>
      </Dialog>
    </Modal>
  </DialogTrigger>

import { Button, Dialog, DialogTrigger, Example, Flexbox, Heading, Modal } from '../../../mdx-components';

A modal is an overlay element which blocks interaction with elements outside it.
Use `Sheet` for panels that slide in from an edge of the viewport.

<Example>
  <DialogTrigger>
    <Button>Center</Button>
    <Modal>
      <Dialog>
        <Flexbox justifyContent="center" alignItems="center" paddingX="9" paddingY="8">
          <Heading>Hello.</Heading>
        </Flexbox>
      </Dialog>
    </Modal>
  </DialogTrigger>
  <DialogTrigger>
    <Button>Fullscreen</Button>
    <Modal placement="fullscreen">
      <Dialog className="size-full rounded-none border-0 shadow-none">
        <Flexbox justifyContent="center" alignItems="center" paddingX="9" paddingY="8">
          <Heading>Hello.</Heading>
        </Flexbox>
      </Dialog>
    </Modal>
  </DialogTrigger>

## Component props

<PropertiesTable>
  <ResponseField name="placement" type="string">"center" | "fullscreen" (default: "center")</ResponseField>
  <ResponseField name="isDismissable" type="boolean">Can be dismissed by clicking outside (default: true)</ResponseField>
  <ResponseField name="isKeyboardDismissDisabled" type="boolean">Disables Escape key dismissal</ResponseField>
</PropertiesTable>

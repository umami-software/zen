import { Checkbox, Example, Table, TableBody, TableCell, TableColumn, TableHeader, TableRow } from '../../../mdx-components';

A table displays data in rows and columns and enables a user to navigate its contents via
directional navigation keys, and optionally supports row selection and sorting.

<Example>
  <Table>
    <TableHeader>
      <TableColumn>Name</TableColumn>
      <TableColumn>Email</TableColumn>
      <TableColumn>Date created</TableColumn>
    </TableHeader>
    <TableBody>
      <TableRow>
        <TableCell>Bob</TableCell>
        <TableCell>bob@aol.com</TableCell>
        <TableCell>5/5/2020</TableCell>

### Table

<PropertiesTable>
  <ResponseField name="selectionMode" type="SelectionMode">"none" | "single" | "multiple"</ResponseField>

  <ResponseField name="defaultSelectedKeys" type="string[]">Default selected row keys</ResponseField>
  <ResponseField name="onSelectionChange" type="(keys: Selection) => void">Called when selection changes</ResponseField>
</PropertiesTable>

<PropertiesTable>

</PropertiesTable>

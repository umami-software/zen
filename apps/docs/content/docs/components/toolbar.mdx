import { Example, Lucide, ToggleGroup, ToggleGroupItem, Toolbar, ToolbarButton, ToolbarGroup, ToolbarSeparator } from '../../../mdx-components';

A toolbar groups a set of buttons and controls with keyboard navigation between them. It is built on
Base UI's `Toolbar`, and `ToolbarButton` takes the shadcn `Button` variants and sizes.

<Example>
  <Toolbar>
    <ToolbarGroup>
      <ToolbarButton size="icon-sm" aria-label="Bold">
        <Lucide.Bold />
      </ToolbarButton>
      <ToolbarButton size="icon-sm" aria-label="Italic">
        <Lucide.Italic />
      </ToolbarButton>
      <ToolbarButton size="icon-sm" aria-label="Underline">
        <Lucide.Underline />
      </ToolbarButton>
    </ToolbarGroup>
    <ToolbarSeparator />
    <ToolbarGroup>
      <ToolbarButton size="icon-sm" aria-label="Align left">
        <Lucide.AlignLeft />
      </ToolbarButton>

        <Lucide.AlignCenter />
      </ToolbarButton>
      <ToolbarButton size="icon-sm" aria-label="Align right">

    </ToolbarGroup>
    <ToolbarSeparator />
    <ToolbarButton size="icon-sm" aria-label="Link" disabled>
      <Lucide.Link />
    </ToolbarButton>
  </Toolbar>

## With toggles

Toolbars work with the shadcn `ToggleGroup` and `ToggleGroupItem` for pressed states.

<Example>
  <Toolbar>
    <ToggleGroup multiple size="sm">
      <ToggleGroupItem value="bold" aria-label="Bold">
        <Lucide.Bold />
      </ToggleGroupItem>
      <ToggleGroupItem value="italic" aria-label="Italic">
        <Lucide.Italic />
      </ToggleGroupItem>
      <ToggleGroupItem value="underline" aria-label="Underline">
        <Lucide.Underline />
      </ToggleGroupItem>
    </ToggleGroup>
    <ToolbarSeparator />
    <ToolbarButton size="icon-sm" aria-label="Undo">
      <Lucide.Undo />
    </ToolbarButton>
    <ToolbarButton size="icon-sm" aria-label="Redo">
      <Lucide.Redo />
    </ToolbarButton>
  </Toolbar>

  ```jsx
  <Toolbar>
    <ToggleGroup multiple size="sm">
      <ToggleGroupItem value="bold" aria-label="Bold">
        <Lucide.Bold />
      </ToggleGroupItem>
      <ToggleGroupItem value="italic" aria-label="Italic">
        <Lucide.Italic />
      </ToggleGroupItem>
      <ToggleGroupItem value="underline" aria-label="Underline">
        <Lucide.Underline />
      </ToggleGroupItem>
    </ToggleGroup>
    <ToolbarSeparator />
    <ToolbarButton size="icon-sm" aria-label="Undo">
      <Lucide.Undo />
    </ToolbarButton>
    <ToolbarButton size="icon-sm" aria-label="Redo">
      <Lucide.Redo />
    </ToolbarButton>
  </Toolbar>
  ```
</Example>

## Component props

### Toolbar

Other Base UI `Toolbar.Root` props (`loopFocus`, …) are passed through.

<PropertiesTable>
  <PropertiesTable.Row name="orientation" type="string">"horizontal" | "vertical" (default: "horizontal")</PropertiesTable.Row>
  <PropertiesTable.Row name="disabled" type="boolean">Disables the entire toolbar</PropertiesTable.Row>
</PropertiesTable>

### ToolbarButton

<PropertiesTable>
  <PropertiesTable.Row name="variant" type="string">shadcn Button variant (default: "ghost")</PropertiesTable.Row>
  <PropertiesTable.Row name="size" type="string">shadcn Button size (default: "sm")</PropertiesTable.Row>
  <PropertiesTable.Row name="disabled" type="boolean">Disables the button</PropertiesTable.Row>
  <PropertiesTable.Row name="onClick" type="(event) => void">Called when the button is clicked</PropertiesTable.Row>
</PropertiesTable>

`ToolbarGroup`, `ToolbarSeparator` and `ToolbarLink` wrap the matching Base UI parts.

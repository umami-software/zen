import { Example, Icon, Lucide, ToggleGroup, ToggleGroupItem, Toolbar, ToolbarButton, ToolbarGroup, ToolbarSeparator } from '../../../mdx-components';

A toolbar groups a set of buttons and controls with keyboard navigation between them.

<Example>
  <Toolbar>
    <ToolbarGroup>
      <ToolbarButton>
        <Icon size="sm"><Lucide.Bold /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><Lucide.Italic /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><Lucide.Underline /></Icon>
      </ToolbarButton>
    </ToolbarGroup>
    <ToolbarSeparator />
    <ToolbarGroup>
      <ToolbarButton>
        <Icon size="sm"><Lucide.AlignLeft /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><Lucide.AlignCenter /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><Lucide.AlignRight /></Icon>
      </ToolbarButton>
    </ToolbarGroup>
    <ToolbarSeparator />
    <ToolbarButton isDisabled>
      <Icon size="sm"><Lucide.Link /></Icon>
    </ToolbarButton>
  </Toolbar>

  ```jsx
  <Toolbar>
    <ToolbarGroup>
      <ToolbarButton>
        <Icon size="sm"><Bold /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><Italic /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><Underline /></Icon>
      </ToolbarButton>
    </ToolbarGroup>
    <ToolbarSeparator />
    <ToolbarGroup>
      <ToolbarButton>
        <Icon size="sm"><AlignLeft /></Icon>
      </ToolbarButton>
      <ToolbarButton>
        <Icon size="sm"><AlignCenter /></Icon>
      </ToolbarButton>
    </ToolbarGroup>
    <ToolbarSeparator />
    <ToolbarButton isDisabled>
      <Icon size="sm"><Link /></Icon>
    </ToolbarButton>
  </Toolbar>
  ```
</Example>

## With toggles

Toolbars work with `ToggleGroup` and `ToggleGroupItem` for pressed states.

<Example>
  <Toolbar>
    <ToggleGroup selectionMode="multiple">
      <ToggleGroupItem id="bold">
        <Icon size="sm"><Lucide.Bold /></Icon>
      </ToggleGroupItem>
      <ToggleGroupItem id="italic">
        <Icon size="sm"><Lucide.Italic /></Icon>
      </ToggleGroupItem>
      <ToggleGroupItem id="underline">
        <Icon size="sm"><Lucide.Underline /></Icon>
      </ToggleGroupItem>
    </ToggleGroup>
    <ToolbarSeparator />
    <ToolbarButton>
      <Icon size="sm"><Lucide.Undo /></Icon>
    </ToolbarButton>
    <ToolbarButton>
      <Icon size="sm"><Lucide.Redo /></Icon>
    </ToolbarButton>
  </Toolbar>

  ```jsx
  <Toolbar>
    <ToggleGroup selectionMode="multiple">
      <ToggleGroupItem id="bold">
        <Icon size="sm"><Bold /></Icon>
      </ToggleGroupItem>
      <ToggleGroupItem id="italic">
        <Icon size="sm"><Italic /></Icon>
      </ToggleGroupItem>
      <ToggleGroupItem id="underline">
        <Icon size="sm"><Underline /></Icon>
      </ToggleGroupItem>
    </ToggleGroup>
    <ToolbarSeparator />
    <ToolbarButton>
      <Icon size="sm"><Undo /></Icon>
    </ToolbarButton>
    <ToolbarButton>
      <Icon size="sm"><Redo /></Icon>
    </ToolbarButton>
  </Toolbar>
  ```
</Example>

## Component props

### Toolbar

<PropertiesTable>
  <ResponseField name="orientation" type="'horizontal' | 'vertical'">Toolbar layout direction (default: "horizontal")</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the entire toolbar</ResponseField>
</PropertiesTable>

### ToolbarButton

<PropertiesTable>
  <ResponseField name="isDisabled" type="boolean">Disables the button</ResponseField>
  <ResponseField name="...ButtonProps" type="ButtonProps">Accepts all Button props (default variant: "quiet")</ResponseField>
</PropertiesTable>

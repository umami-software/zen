import { Box, Example, ThemeButton, ThemeButtonExample } from '../../../mdx-components';

A button that toggles the page theme between light and dark.

<Example>
  <ThemeButton variant="outline" />

  ```jsx
  <ThemeButton variant="outline" />
  ```
</Example>

## Local target

Use the `target` prop to scope theme changes to a specific container instead of globally.

<Example>
  <ThemeButtonExample />

  ```jsx
  const ref = useRef(null);

  <Box ref={ref}>
    <ThemeButton target={ref} />
  </Box>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="target" type="RefObject | HTMLElement">Element to apply theme to (default: global)</ResponseField>
  <ResponseField name="variant" type="ButtonVariant">"default" | "primary" | "outline" | "quiet" (default: "quiet")</ResponseField>
  <ResponseField name="size" type="ButtonSize">"xs" | "sm" | "md" | "lg" | "xl"</ResponseField>
  <ResponseField name="onPress" type="(e: PressEvent) => void">Called when button is pressed</ResponseField>
</PropertiesTable>

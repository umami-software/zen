import { AlertDialog, AlertDialogExample, Button, DialogTrigger, Example, Modal, Text } from '../../../mdx-components';

Alert dialogs are a specific type of Dialog. They display important information that users need
to acknowledge.

<Example>
  <AlertDialogExample />

  ```jsx
    <DialogTrigger overlayType="alert-dialog">
    <Button variant="primary">Delete account</Button>
    <Modal>
      <AlertDialog title="Are you sure you want to delete your account?">
        <Text>This cannot be undone. All your data will be permanantly deleted.</Text>
      </AlertDialog>
    </Modal>
  </DialogTrigger>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="title" type="ReactNode">Dialog title</ResponseField>
  <ResponseField name="description" type="ReactNode">Dialog description</ResponseField>
  <ResponseField name="isDanger" type="boolean">Shows confirm button as danger variant</ResponseField>
  <ResponseField name="isConfirmDisabled" type="boolean">Disables the confirm button</ResponseField>
  <ResponseField name="confirmLabel" type="ReactNode">Confirm button text (default: "Confirm")</ResponseField>
  <ResponseField name="cancelLabel" type="ReactNode">Cancel button text (default: "Cancel")</ResponseField>
  <ResponseField name="onConfirm" type="() => void | Promise<unknown>">Called when confirm is clicked. When it returns a promise the dialog stays open until it resolves, and stays open on rejection</ResponseField>
  <ResponseField name="onCancel" type="() => void">Called when cancel is clicked</ResponseField>
</PropertiesTable>

A sheet is a panel that slides in from the edge of the screen, overlaying the current content.

<Example>
  <SheetExample />

  ```jsx
  <DialogTrigger>
    <Button variant="outline">Open sheet</Button>
    <Sheet side="right">
      <SheetHeader title="Settings" />
      <Text>Sheet content slides in from the right.</Text>
    </Sheet>
  </DialogTrigger>
  ```
</Example>

## Size

The `size` prop sets the width for left and right sheets, and the height for top and bottom sheets.

<Example>
  <DialogTrigger>
    <Button variant="outline">Wide sheet</Button>
    <Sheet side="right" size="40rem">
      <SheetHeader title="Details" />
      <Text>A wider sheet panel.</Text>
    </Sheet>
  </DialogTrigger>

  ```jsx
  <DialogTrigger>
    <Button variant="outline">Wide sheet</Button>
    <Sheet side="right" size="40rem">
      <SheetHeader title="Details" />
      <Text>A wider sheet panel.</Text>
    </Sheet>
  </DialogTrigger>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="side" type="string">"left" | "right" | "top" | "bottom" (default: "right")</ResponseField>
  <ResponseField name="size" type="string | number">Width (left/right) or height (top/bottom) of the sheet</ResponseField>
  <ResponseField name="isOpen" type="boolean">Controlled open state</ResponseField>
  <ResponseField name="defaultOpen" type="boolean">Initial open state. Used when the sheet is not wrapped in a `DialogTrigger`</ResponseField>
  <ResponseField name="onOpenChange" type="(open: boolean) => void">Called when the open state changes</ResponseField>
  <ResponseField name="showCloseButton" type="boolean">Renders a close button in the top-right corner. Defaults to `true` unless a `SheetHeader` is present, so only one close button is ever rendered</ResponseField>
</PropertiesTable>

## Parts

<PropertiesTable>
  <ResponseField name="SheetTrigger" type="component">Opens the sheet</ResponseField>
  <ResponseField name="SheetClose" type="component">Closes the sheet</ResponseField>
  <ResponseField name="SheetTitle" type="component">Accessible title, rendered as a `Heading`</ResponseField>
  <ResponseField name="SheetDescription" type="component">Accessible description, rendered as muted `Text`</ResponseField>
</PropertiesTable>

## SheetHeader props

<PropertiesTable>
  <ResponseField name="title" type="ReactNode">Title displayed at the top of the sheet</ResponseField>
  <ResponseField name="showClose" type="boolean">Shows the close button (default: true)</ResponseField>
  <ResponseField name="children" type="ReactNode">Additional content below the title</ResponseField>
</PropertiesTable>

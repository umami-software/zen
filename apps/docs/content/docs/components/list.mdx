import { Example, List, ListItem, ListSection, ListSeparator } from '../../../mdx-components';

A list displays a collection of items and allows a user to select one or more of them.

<Example>
  <List>
    <ListItem>One</ListItem>
    <ListItem>Two</ListItem>
    <ListItem>Three</ListItem>
    <ListItem>Four</ListItem>
    <ListItem>Five</ListItem>
  </List>

  ```jsx
  <List>
    <ListItem>One</ListItem>
    <ListItem>Two</ListItem>
    <ListItem>Three</ListItem>
  </List>
  ```
</Example>

## Single select

<Example>
  <List selectionMode="single">
    <ListItem>One</ListItem>
    <ListItem>Two</ListItem>
    <ListItem>Three</ListItem>
    <ListItem>Four</ListItem>
    <ListItem>Five</ListItem>
  </List>

  ```jsx
  <List selectionMode="single">
    <ListItem>One</ListItem>
    <ListItem>Two</ListItem>
    <ListItem>Three</ListItem>
  </List>
  ```
</Example>

## Multiple select

<Example>
  <List selectionMode="multiple">
    <ListItem>One</ListItem>
    <ListItem>Two</ListItem>
    <ListItem>Three</ListItem>
    <ListItem>Four</ListItem>
    <ListItem>Five</ListItem>
  </List>

  ```jsx
  <List selectionMode="multiple">
    <ListItem>One</ListItem>
    <ListItem>Two</ListItem>
    <ListItem>Three</ListItem>
  </List>
  ```
</Example>

## Preselected

<Example>
  <List selectionMode="multiple" defaultSelectedKeys={['three', 'five']}>
    <ListItem id="one">One</ListItem>
    <ListItem id="two">Two</ListItem>
    <ListItem id="three">Three</ListItem>
    <ListItem id="four">Four</ListItem>
    <ListItem id="five">Five</ListItem>
  </List>

  ```jsx
  <List selectionMode="multiple" defaultSelectedKeys={['three', 'five']}>
    <ListItem id="one">One</ListItem>
    <ListItem id="two">Two</ListItem>
    <ListItem id="three">Three</ListItem>
  </List>
  ```
</Example>

## Component props

### List

<PropertiesTable>
  <ResponseField name="selectionMode" type="SelectionMode">"none" | "single" | "multiple"</ResponseField>
  <ResponseField name="selectedKeys" type="string[]">Controlled selected keys</ResponseField>
  <ResponseField name="defaultSelectedKeys" type="string[]">Default selected keys</ResponseField>
  <ResponseField name="onSelectionChange" type="(keys: Selection) => void">Called when selection changes</ResponseField>
</PropertiesTable>

### ListItem

<PropertiesTable>
  <ResponseField name="id" type="string">Unique item identifier</ResponseField>
  <ResponseField name="textValue" type="string">Text value for accessibility</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the item</ResponseField>
</PropertiesTable>

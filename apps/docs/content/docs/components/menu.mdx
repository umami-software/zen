  ```jsx

  </Menu>
  ```

</Example>

## Submenu

<Example>
  <MenuTrigger>
    <Button variant="outline">Open menu</Button>
    <Menu>
      <MenuItem id="one">One</MenuItem>
      <MenuItem id="two">Two</MenuItem>
      <MenuItem id="three">Three</MenuItem>
      <MenuItem id="four">Four</MenuItem>
      <SubMenuTrigger>
        <MenuItem id="five" showSubMenuIcon>Five</MenuItem>
        <Menu>
          <MenuItem id="six">Six</MenuItem>
          <MenuItem id="seven">Seven</MenuItem>
          <MenuItem id="eight">Eight</MenuItem>
        </Menu>
      </SubMenuTrigger>
    </Menu>
  </MenuTrigger>

  ```jsx
  <MenuTrigger>
    <Button>Open menu</Button>
    <Menu>
      <MenuItem>One</MenuItem>
      <SubMenuTrigger>
        <MenuItem showSubMenuIcon>More</MenuItem>
        <Menu>
          <MenuItem>Sub item 1</MenuItem>
          <MenuItem>Sub item 2</MenuItem>
        </Menu>
      </SubMenuTrigger>
    </Menu>
  </MenuTrigger>
  ```
</Example>

## Component props

### Menu

<PropertiesTable>
  <ResponseField name="selectionMode" type="SelectionMode">"none" | "single" | "multiple"</ResponseField>
  <ResponseField name="selectedKeys" type="string[]">Controlled selected keys</ResponseField>
  <ResponseField name="defaultSelectedKeys" type="string[]">Default selected keys</ResponseField>
  <ResponseField name="onSelectionChange" type="(keys: Selection) => void">Called when selection changes</ResponseField>
  <ResponseField name="variant" type="'default' | 'plain'">`plain` drops the popup surface (border, shadow, background) when the menu is embedded in another popup, such as a `Navbar` dropdown</ResponseField>
</PropertiesTable>

With `selectionMode="single"` items render as radio items and with `"multiple"` as checkbox items, so
`aria-checked` is exposed to assistive technology.

### MenuItem

<PropertiesTable>
  <ResponseField name="id" type="string">Unique item identifier</ResponseField>
  <ResponseField name="icon" type="ReactNode">Icon displayed before label</ResponseField>
  <ResponseField name="label" type="string">Item label text</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the item</ResponseField>
  <ResponseField name="showChecked" type="boolean">Shows the selected indicator. Defaults to `true` only when the menu has a selection mode</ResponseField>
  <ResponseField name="showSubMenuIcon" type="boolean">Shows submenu arrow icon</ResponseField>
  <ResponseField name="closeOnSelect" type="boolean">Whether picking the item closes the menu. Defaults to `false` for multiple selection</ResponseField>
</PropertiesTable>

### MenuShortcut

Renders a right-aligned keyboard hint inside a `MenuItem`.

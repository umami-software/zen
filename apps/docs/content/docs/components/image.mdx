import { Box, Column, Example, Image, Label } from '../../../mdx-components';

An image component with built-in support for responsive sizing, object-fit modes, and border radius. Handles common image display patterns while maintaining aspect ratios and proper scaling.

<Example>
  <Box maxWidth="480px">
    <Image src="/images/sample.jpg" alt="Image" />
  </Box>

  ```jsx
  <Image src="/images/sample.jpg" alt="Image" />
  ```
</Example>

## Object fit

<Example>
  <Column gap="4">
    <Label>Cover</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="cover" />
    </Box>
    <Label>Fill</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="fill" />
    </Box>
    <Label>Contain</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="contain" />
    </Box>
    <Label>Scale down</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="scale-down" />
    </Box>
    <Label>None</Label>
    <Box maxWidth="480px" height="480px" border overflow="hidden">
      <Image src="/images/sample.jpg" alt="Image" objectFit="none" />
    </Box>
  </Column>

  ```jsx
  <Image src="/images/sample.jpg" alt="Image" objectFit="cover" />
  <Image src="/images/sample.jpg" alt="Image" objectFit="contain" />
  <Image src="/images/sample.jpg" alt="Image" objectFit="fill" />
  ```
</Example>

## Centered and rounded

<Example>
  <Box maxWidth="640px">
    <Image src="/images/sample.jpg" alt="Image" centered borderRadius="xl" shadow="md" style={{ width: 320 }} />
  </Box>

  ```jsx
  <Image src="/images/sample.jpg" alt="Image" centered borderRadius="xl" shadow="md" style={{ width: 320 }} />
  ```

</Example>

## Component props

<PropertiesTable>
  <PropertiesTable.Row name="src" type="string">Image source URL (required)</PropertiesTable.Row>
  <PropertiesTable.Row name="alt" type="string">Alternative text</PropertiesTable.Row>
  <PropertiesTable.Row name="objectFit" type="ObjectFit">"fill" | "contain" | "cover" | "none" | "scale-down"</PropertiesTable.Row>
  <PropertiesTable.Row name="centered" type="boolean">Centers the image horizontally</PropertiesTable.Row>
  <PropertiesTable.Row name="borderRadius" type="BorderRadius">Border radius value</PropertiesTable.Row>
  <PropertiesTable.Row name="shadow" type="BoxShadow">Box shadow value</PropertiesTable.Row>
</PropertiesTable>

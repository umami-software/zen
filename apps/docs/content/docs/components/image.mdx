title: Image

import { Box, Column, Example, Image, Label } from '../../../mdx-components';

An image component with built-in support for responsive sizing, object-fit modes, and border radius. Handles common image display patterns while maintaining aspect ratios and proper scaling.

<Example>
  <Box maxWidth="480px">
    <Image src="/images/sample.jpg" alt="Image" />
  </Box>

  ```jsx
  <Image src="/images/sample.jpg" alt="Image" />
  ```
</Example>

## Object fit

<Example>
  <Column gap="4">
    <Label>Cover</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="cover" />
    </Box>
    <Label>Fill</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="fill" />
    </Box>
    <Label>Contain</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="contain" />
    </Box>
    <Label>Scale down</Label>
    <Box maxWidth="480px" height="480px" border>
      <Image src="/images/sample.jpg" alt="Image" objectFit="scale-down" />
    </Box>
    <Label>None</Label>
    <Box maxWidth="480px" height="480px" border overflow="hidden">
      <Image src="/images/sample.jpg" alt="Image" objectFit="none" />
    </Box>
  </Column>

  ```jsx
  <Image src="/images/sample.jpg" alt="Image" objectFit="cover" />
  <Image src="/images/sample.jpg" alt="Image" objectFit="contain" />
  <Image src="/images/sample.jpg" alt="Image" objectFit="fill" />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="src" type="string">Image source URL (required)</ResponseField>
  <ResponseField name="alt" type="string">Alternative text</ResponseField>
  <ResponseField name="objectFit" type="ObjectFit">"fill" | "contain" | "cover" | "none" | "scale-down"</ResponseField>
  <ResponseField name="isCentered" type="boolean">Centers the image horizontally</ResponseField>
  <ResponseField name="borderRadius" type="BorderRadius">Border radius value</ResponseField>
  <ResponseField name="shadow" type="BoxShadow">Box shadow value</ResponseField>
</PropertiesTable>

import { Column, Example, Pagination, PaginationExample, Text } from '../../../mdx-components';

Pagination lets users navigate between pages of content.

<Example>
  <Pagination pageCount={10} defaultPage={1} />

  ```jsx
  <Pagination pageCount={10} defaultPage={1} />
  ```
</Example>

## Total items

Instead of `pageCount`, pass `totalItems` and `pageSize` and the page count is calculated automatically.

<Example>
  <Pagination totalItems={250} pageSize={25} />

  ```jsx
  <Pagination totalItems={250} pageSize={25} />
  ```
</Example>

## Sibling count

Control how many pages are shown on each side of the current page with `siblingCount`.

<Example>
  <Pagination pageCount={20} defaultPage={10} siblingCount={2} />

  ```jsx
  <Pagination pageCount={20} defaultPage={10} siblingCount={2} />
  ```
</Example>

## Controlled

Use the `page` and `onChange` props to control the current page.

<Example>
  <PaginationExample />

  ```jsx
  function App() {
    const [page, setPage] = useState(1);

    return (
      <Column gap="3" alignItems="center">
        <Pagination pageCount={10} page={page} onChange={setPage} />
        <Text color="muted">Current page: {page}</Text>
      </Column>
    );
  }
  ```
</Example>

## Disabled

<Example>
  <Pagination pageCount={10} defaultPage={3} isDisabled />

  ```jsx
  <Pagination pageCount={10} defaultPage={3} isDisabled />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="pageCount" type="number">Total number of pages</ResponseField>
  <ResponseField name="totalItems" type="number">Total number of items (used when pageCount is not set)</ResponseField>
  <ResponseField name="pageSize" type="number">Items per page used with totalItems (default: 10)</ResponseField>
  <ResponseField name="page" type="number">Controlled current page</ResponseField>
  <ResponseField name="defaultPage" type="number">Default current page (default: 1)</ResponseField>
  <ResponseField name="siblingCount" type="number">Pages shown on each side of the current page (default: 1)</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables all pagination controls</ResponseField>
  <ResponseField name="onChange" type="(page: number) => void">Called when the page changes</ResponseField>
</PropertiesTable>

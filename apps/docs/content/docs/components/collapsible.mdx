import { Box, Collapsible, CollapsiblePanel, CollapsibleTrigger, Example, Text } from '../../../mdx-components';

A collapsible shows and hides a section of content.

<Example>
  <Box width="300px">
    <Collapsible>
      <CollapsibleTrigger>Show details</CollapsibleTrigger>
      <CollapsiblePanel>
        <Box paddingY="2">
          <Text>Here are the details you were looking for.</Text>
        </Box>
      </CollapsiblePanel>
    </Collapsible>
  </Box>

  ```jsx
  <Collapsible>
    <CollapsibleTrigger>Show details</CollapsibleTrigger>
    <CollapsiblePanel>
      <Text>Here are the details you were looking for.</Text>
    </CollapsiblePanel>
  </Collapsible>
  ```
</Example>

## Default expanded

<Example>
  <Box width="300px">
    <Collapsible defaultExpanded>
      <CollapsibleTrigger>Show details</CollapsibleTrigger>
      <CollapsiblePanel>
        <Box paddingY="2">
          <Text>This panel starts out expanded.</Text>
        </Box>
      </CollapsiblePanel>
    </Collapsible>
  </Box>

  ```jsx
  <Collapsible defaultExpanded>
    <CollapsibleTrigger>Show details</CollapsibleTrigger>
    <CollapsiblePanel>
      <Text>This panel starts out expanded.</Text>
    </CollapsiblePanel>
  </Collapsible>
  ```
</Example>

## Component props

### Collapsible

<PropertiesTable>
  <ResponseField name="isExpanded" type="boolean">Controlled expanded state</ResponseField>
  <ResponseField name="defaultExpanded" type="boolean">Default expanded state</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the collapsible</ResponseField>
  <ResponseField name="onExpandedChange" type="(expanded: boolean) => void">Called when the expanded state changes</ResponseField>
</PropertiesTable>

### CollapsibleTrigger

The button that toggles the panel.

### CollapsiblePanel

The content that is shown and hidden.

import { Example, ProgressCircle } from '../../../mdx-components';

A circular progress indicator that visually represents completion status. Ideal for compact spaces, dashboards, or when you want a more visually distinctive progress display than a bar.

<Example>
  <ProgressCircle value={45} />

  ```jsx
  <ProgressCircle value={45} />
  ```
</Example>

## Show percentage

<Example>
  <ProgressCircle value={15} showPercentage />

  ```jsx
  <ProgressCircle value={15} showPercentage />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="value" type="number">Current progress value</ResponseField>
  <ResponseField name="minValue" type="number">Minimum value (default: 0)</ResponseField>
  <ResponseField name="maxValue" type="number">Maximum value (default: 100)</ResponseField>
  <ResponseField name="showPercentage" type="boolean">Shows percentage text in center</ResponseField>
</PropertiesTable>

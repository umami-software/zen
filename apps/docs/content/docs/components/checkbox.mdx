title: Checkbox

import { Checkbox, Example } from '../../../mdx-components';

A checkbox allows a user to select multiple items from a list of individual items, or to mark one individual item as selected.

<Example>
  <Checkbox>Accept terms and conditions.</Checkbox>

  ```jsx
  <Checkbox>Accept terms and conditions.</Checkbox>
  ```
</Example>

## Indeterminate

<Example>
  <Checkbox isIndeterminate={true}>Accept terms and conditions.</Checkbox>

  ```jsx
  <Checkbox isIndeterminate>Accept terms and conditions.</Checkbox>
  ```
</Example>

## Disabled

<Example>
  <Checkbox isDisabled={true}>Accept terms and conditions.</Checkbox>

  ```jsx
  <Checkbox isDisabled>Accept terms and conditions.</Checkbox>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text (alternative to children)</ResponseField>
  <ResponseField name="isSelected" type="boolean">Controlled selected state</ResponseField>
  <ResponseField name="defaultSelected" type="boolean">Default selected state</ResponseField>
  <ResponseField name="isIndeterminate" type="boolean">Shows indeterminate state</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the checkbox</ResponseField>
  <ResponseField name="onChange" type="(isSelected: boolean) => void">Called when state changes</ResponseField>
</PropertiesTable>

import { Button, Example, FloatingTooltip, FloatingTooltipExample, Tooltip, TooltipTrigger } from '../../../mdx-components';

A tooltip displays a description of an element on hover or focus.

<Example>
  <TooltipTrigger delay={0}>
    <Button variant="secondary">Hover me</Button>
    <Tooltip>Hello</Tooltip>
  </TooltipTrigger>

  ```jsx
  <TooltipTrigger delay={0}>
    <Button variant="secondary">Hover me</Button>
    <Tooltip>Hello</Tooltip>
  </TooltipTrigger>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="placement" type="Placement">"top" | "bottom" | "left" | "right"</ResponseField>
  <ResponseField name="showArrow" type="boolean">Shows an arrow pointing to the trigger</ResponseField>
  <ResponseField name="offset" type="number">Distance from the trigger element</ResponseField>
</PropertiesTable>

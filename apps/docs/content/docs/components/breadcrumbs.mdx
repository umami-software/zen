---
title: Breadcrumbs

import { Breadcrumb, Breadcrumbs, Example } from '../../../mdx-components';

Breadcrumbs display a hierarchy of links to the current page or resource in an application.

<Example>
  <Breadcrumbs>
    <Breadcrumb>
      <a href="#">Home</a>
    </Breadcrumb>
    <Breadcrumb>
      <a href="#">One</a>
    </Breadcrumb>
    <Breadcrumb>
      <a href="#">Two</a>
    </Breadcrumb>
    <Breadcrumb>Three</Breadcrumb>
  </Breadcrumbs>

  ```jsx
  <Breadcrumbs>
    <Breadcrumb><a href="#">Home</a></Breadcrumb>
    <Breadcrumb><a href="#">One</a></Breadcrumb>
    <Breadcrumb><a href="#">Two</a></Breadcrumb>
    <Breadcrumb>Three</Breadcrumb>
  </Breadcrumbs>
  ```
</Example>

## Component props

### Breadcrumbs

<PropertiesTable>
  <ResponseField name="isDisabled" type="boolean">Disables all breadcrumbs</ResponseField>
  <ResponseField name="onAction" type="(key: string) => void">Called when a breadcrumb is clicked</ResponseField>
</PropertiesTable>

### Breadcrumb

<PropertiesTable>
  <ResponseField name="id" type="string">Unique breadcrumb identifier</ResponseField>
  <ResponseField name="href" type="string">Link URL</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the breadcrumb</ResponseField>
</PropertiesTable>

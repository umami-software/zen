import { Checkbox, CheckboxGroup, CheckboxGroupExample, Example, Field, FieldLabel } from '../../../mdx-components';

A checkbox group allows a user to select multiple items from a list of options. It wraps Base UI's
`CheckboxGroup` in a fieldset with an optional legend. Each `Checkbox` needs a `value` and its own
label, using `Field` and `FieldLabel`.

<Example>
  <CheckboxGroup label="Notifications" defaultValue={['email']}>
    <Field orientation="horizontal">
      <Checkbox id="cg-email" value="email" />
      <FieldLabel htmlFor="cg-email">Email</FieldLabel>
    </Field>
    <Field orientation="horizontal">
      <Checkbox id="cg-sms" value="sms" />
      <FieldLabel htmlFor="cg-sms">SMS</FieldLabel>
    </Field>
    <Field orientation="horizontal">
      <Checkbox id="cg-push" value="push" />
      <FieldLabel htmlFor="cg-push">Push</FieldLabel>
    </Field>
  </CheckboxGroup>

## Controlled

<Example>
  <CheckboxGroupExample />

  ```jsx
  const [value, setValue] = useState(['email']);

  <CheckboxGroup label="Notifications" value={value} onValueChange={setValue}>
    <Field orientation="horizontal">
      <Checkbox id="notify-email" value="email" />
      <FieldLabel htmlFor="notify-email">Email</FieldLabel>
    </Field>
    <Field orientation="horizontal">
      <Checkbox id="notify-sms" value="sms" />
      <FieldLabel htmlFor="notify-sms">SMS</FieldLabel>
    </Field>
    <Field orientation="horizontal">
      <Checkbox id="notify-push" value="push" />
      <FieldLabel htmlFor="notify-push">Push</FieldLabel>
    </Field>
  </CheckboxGroup>
  <Text color="muted">Selected: {value.length ? value.join(', ') : '(none)'}</Text>
  ```
</Example>

## Component props

Other Base UI `CheckboxGroup` props (`allValues`, `name`) are passed through.

  <PropertiesTable.Row name="label" type="ReactNode">Legend displayed above the group</PropertiesTable.Row>

  <PropertiesTable.Row name="value" type="string[]">Controlled selected values</PropertiesTable.Row>
  <PropertiesTable.Row name="defaultValue" type="string[]">Initial selected values</PropertiesTable.Row>
  <PropertiesTable.Row name="onValueChange" type="(value: string[], details) => void">Called when the selection changes</PropertiesTable.Row>

  <PropertiesTable.Row name="disabled" type="boolean">Disables all checkboxes in the group</PropertiesTable.Row>
  <PropertiesTable.Row name="children" type="ReactNode">Checkboxes with `value` props, each with its own label</PropertiesTable.Row>
</PropertiesTable>

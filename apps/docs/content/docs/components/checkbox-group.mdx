import { Checkbox, CheckboxGroup, CheckboxGroupExample, Example } from '../../../mdx-components';

A checkbox group allows a user to select multiple items from a list of options.

<Example>
  <CheckboxGroup label="Notifications" defaultValue={['email']}>
    <Checkbox value="email">Email</Checkbox>
    <Checkbox value="sms">SMS</Checkbox>
    <Checkbox value="push">Push</Checkbox>
  </CheckboxGroup>

  ```jsx
  <CheckboxGroup label="Notifications" defaultValue={['email']}>
    <Checkbox value="email">Email</Checkbox>
    <Checkbox value="sms">SMS</Checkbox>
    <Checkbox value="push">Push</Checkbox>
  </CheckboxGroup>
  ```
</Example>

## Controlled

<Example>
  <CheckboxGroupExample />

  ```jsx
  const [value, setValue] = useState(['email']);

  <CheckboxGroup label="Notifications" value={value} onChange={setValue}>
    <Checkbox value="email">Email</Checkbox>
    <Checkbox value="sms">SMS</Checkbox>
    <Checkbox value="push">Push</Checkbox>
  </CheckboxGroup>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the group</ResponseField>
  <ResponseField name="value" type="string[]">Controlled selected values</ResponseField>
  <ResponseField name="defaultValue" type="string[]">Default selected values</ResponseField>
  <ResponseField name="orientation" type="string">"vertical" | "horizontal" (default: "vertical")</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables all checkboxes in the group</ResponseField>
  <ResponseField name="onChange" type="(value: string[]) => void">Called when the selection changes</ResponseField>
  <ResponseField name="children" type="ReactNode">Checkbox components with value props</ResponseField>
</PropertiesTable>

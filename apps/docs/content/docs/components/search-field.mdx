## Component props

<PropertiesTable>
  <ResponseField name="label" type="string">Label text displayed above the field</ResponseField>
  <ResponseField name="placeholder" type="string">Placeholder text</ResponseField>
  <ResponseField name="value" type="string">Controlled value</ResponseField>
  <ResponseField name="defaultValue" type="string">Default value (uncontrolled)</ResponseField>
  <ResponseField name="delay" type="number">Debounce delay in ms before onSearch fires (default: 0)</ResponseField>
  <ResponseField name="onChange" type="(value: string) => void">Called on every change</ResponseField>
  <ResponseField name="onSearch" type="(value: string) => void">Called after delay (debounced)</ResponseField>
</PropertiesTable>

An OTP field allows a user to enter a one-time password or verification code, one character per input slot.

<Example>
  <OTPField label="Verification code" />

  ```jsx
  <OTPField label="Verification code" />
  ```
</Example>

## Controlled

<Example>
  <OTPFieldExample />

  ```jsx
  const [value, setValue] = useState('');
  const [isComplete, setIsComplete] = useState(false);

  <OTPField
    label="Verification code"
    value={value}
    onChange={setValue}
    onComplete={() => setIsComplete(true)}
  />
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="length" type="number">Number of input slots (default: 6)</ResponseField>
  <ResponseField name="label" type="string">Label text displayed above the field</ResponseField>
  <ResponseField name="value" type="string">Controlled value</ResponseField>
  <ResponseField name="defaultValue" type="string">Default value</ResponseField>
  <ResponseField name="mask" type="boolean">Masks entered characters (default: false)</ResponseField>
  <ResponseField name="autoSubmit" type="boolean">Submits the owning form when all slots are filled</ResponseField>
  <ResponseField name="validationType" type="string">"numeric" | "alpha" | "alphanumeric" | "none"</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the field</ResponseField>
  <ResponseField name="isReadOnly" type="boolean">Prevents changing the value</ResponseField>
  <ResponseField name="isRequired" type="boolean">Marks the field as required</ResponseField>
  <ResponseField name="onChange" type="(value: string) => void">Called when the value changes</ResponseField>
  <ResponseField name="onComplete" type="(value: string) => void">Called when all slots are filled</ResponseField>
</PropertiesTable>

---
title: Menubar
---

import { Example, Menu, MenuItem, MenuSeparator, Menubar, MenubarMenu } from '../../../mdx-components';

A menubar displays a horizontal row of menus, like the menu bar found at the top of desktop applications.

<Example>
  <Menubar>
    <MenubarMenu label="File">
      <Menu>
        <MenuItem>New file</MenuItem>
        <MenuItem>Open</MenuItem>
        <MenuSeparator />
        <MenuItem>Save</MenuItem>
        <MenuItem>Save as</MenuItem>
      </Menu>
    </MenubarMenu>
    <MenubarMenu label="Edit">
      <Menu>
        <MenuItem>Undo</MenuItem>
        <MenuItem>Redo</MenuItem>
        <MenuSeparator />
        <MenuItem>Cut</MenuItem>
        <MenuItem>Copy</MenuItem>
        <MenuItem>Paste</MenuItem>
      </Menu>
    </MenubarMenu>
    <MenubarMenu label="View">
      <Menu>
        <MenuItem>Zoom in</MenuItem>
        <MenuItem>Zoom out</MenuItem>
        <MenuSeparator />
        <MenuItem>Full screen</MenuItem>
      </Menu>
    </MenubarMenu>
    <MenubarMenu label="Help" isDisabled>
      <Menu>
        <MenuItem>Documentation</MenuItem>
      </Menu>
    </MenubarMenu>
  </Menubar>


</Example>

## Component props

### Menubar

<PropertiesTable>
  <ResponseField name="isDisabled" type="boolean">Disables the entire menubar</ResponseField>
</PropertiesTable>

### MenubarMenu

<PropertiesTable>
  <ResponseField name="label" type="ReactNode">Label shown on the menubar trigger button</ResponseField>
  <ResponseField name="isDisabled" type="boolean">Disables the menu</ResponseField>
  <ResponseField name="children" type="ReactNode">A Menu containing MenuItems</ResponseField>
</PropertiesTable>

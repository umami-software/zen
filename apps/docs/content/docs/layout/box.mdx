import { Box, Example } from '../../../mdx-components';

A versatile layout primitive that serves as the foundation for building UI. Box provides comprehensive styling props for spacing, colors, borders, positioning, and more. Use it to wrap and position other components, create cards, or build custom layouts.

<Example>
  <Box>I am a box.</Box>

  ```jsx
  <Box>I am a box.</Box>
  ```
</Example>

## With properties

<Example>
  <Box padding="5" shadow="lg" borderRadius="lg" border>
    I am a box.
  </Box>

  ```jsx
  <Box padding="5" shadow="lg" borderRadius="lg" border>
    I am a box.
  </Box>
  ```
</Example>

## Responsive sizing

Use Tailwind preset values for width and height with responsive breakpoints. Resize your browser window to see this box change width: full width on mobile, half width on medium screens, and one-third on large screens.

<Example alignItems="stretch">
  <Box width={{ base: 'full', md: '1/2', lg: '1/3' }} padding="4" backgroundColor="surface-raised" border borderRadius>
    Responsive width
  </Box>

  ```jsx
  <Box
    width={{ base: 'full', md: '1/2', lg: '1/3' }}
    padding="4"
    backgroundColor="surface-raised"
    border
    borderRadius>
    Responsive width
  </Box>
  ```
</Example>

## State styles

Apply styles for hover, focus, and active states using object props. Each state accepts `color`, `backgroundColor`, `borderColor`, and `opacity`.

<Example>
  <Box
    padding="4"
    border
    borderRadius
    cursor="pointer"
    hover={{ color: 'red-600', backgroundColor: "red-100", borderColor: "red-500" }}
  >
    Hover over me
  </Box>

  ```jsx
  <Box
    padding="4"
    border
    borderRadius
    cursor="pointer"
    hover={{
      color: 'red-600',
      backgroundColor: "red-100",
      borderColor: "red-500"
    }}>
    Hover over me
  </Box>
  ```
</Example>

### Multiple states

Click to focus, then hover and click to see all states.

<Example>
  <Box
    tabIndex={0}
    padding="4"
    backgroundColor="surface-base"
    border
    borderRadius
    cursor="pointer"
    hover={{ backgroundColor: "blue-50" }}
    focus={{ borderColor: "blue-500" }}
    active={{ backgroundColor: "blue-100", opacity: "50" }}
  >
    Interactive box (hover, focus, active)
  </Box>

  ```jsx
  <Box
    tabIndex={0}
    padding="4"
    backgroundColor="surface-base"
    border
    borderRadius
    cursor="pointer"
    hover={{ backgroundColor: "blue-50" }}
    focus={{ borderColor: "blue-500" }}
    active={{ backgroundColor: "blue-100", opacity: "50" }}
  >
    Interactive box (hover, focus, active)
  </Box>
  ```
</Example>

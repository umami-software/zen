import { Box, Example, ScrollArea } from '../../../mdx-components';

A scroll area provides scrollable content with custom styled scrollbars.

<Example>
  <Box width="300px" border borderRadius="lg" padding="2">
    <ScrollArea maxHeight={200}>
      {Array.from({ length: 20 }, (_, i) => (
        <Box key={i} paddingX="3" paddingY="2">Row {i + 1}</Box>
      ))}
    </ScrollArea>
  </Box>

  ```jsx
  <ScrollArea maxHeight={200}>
    {Array.from({ length: 20 }, (_, i) => (
      <Box key={i} paddingX="3" paddingY="2">Row {i + 1}</Box>
    ))}
  </ScrollArea>
  ```
</Example>

## Component props

<PropertiesTable>
  <ResponseField name="maxHeight" type="string | number">Maximum height of the scrollable viewport</ResponseField>
  <ResponseField name="orientation" type="string">"vertical" | "horizontal" | "both" (default: "vertical")</ResponseField>
</PropertiesTable>
